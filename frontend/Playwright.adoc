= E2E Test: Planungs-Workflow mit Playwright

Ich habe einen End-to-End (E2E) Test mit Playwright erstellt, der einen typischen Planungsablauf in der Anwendung simuliert. Dieser Test dient dazu, den kompletten Workflow von der Anlage einer Veranstaltung bis zur Vorbereitung der Optimierung zu verifizieren.

== Testfall: `PlanungsWorkflow.spec.js`

Der Test simuliert die folgenden Schritte im Frontend:

. **Login** als Administrator (`juergenkrey@yahoo.de` mit Passwort `start123`).
. **Anlegen einer neuen Veranstaltung**:
** Name: "Test Event 2025"
** Beginn: "2025-05-20T09:00"
** Ort: "Linz am Rhein"
. **Auswahl dieser Veranstaltung** im Dropdown-Menü des Admin-Dashboards.
. **Anlegen eines Gebäudes**:
** Name: "Hauptgebäude A"
** Adresse: "Schulstraße 1, 53545 Linz"
. **Anlegen von 3 EventSlots**:
** Slot 1: 09:00 - 09:45
** Slot 2: 10:00 - 10:45
** Slot 3: 11:00 - 11:45
. **Anlegen eines Referenten**:
** Name: "Max Referent"
** E-Mail: "max@referent.de"
. **Anlegen eines Vortrags** und Zuweisung zum eben erstellten Referenten.
. **Anlegen eines Teilnehmers**:
** Name: "Tom Teilnehmer"
** E-Mail: "tom@student.de"
. **Navigation zum Tab "Vortragsplanung"**.
. **Verifikation der angezeigten Zählwerte** für Teilnehmer, Vorträge und Referenten in der Übersichtsbox.
. **Prüfung, ob der "MiniZinc Optimierung starten"-Button aktiv ist**.

=== Testcode: `frontend/tests/PlanungsWorkflow.spec.js`

[source,javascript]
----
import { test, expect } from '@playwright/test';

test.describe('Vortragsmanager Planungs-Workflow', () => {

  test('Vollständiger Workflow: Veranstaltung -> Gebäude -> Slots -> Personen -> Vorträge', async ({ page }) => {
    // 1. LOGIN
    await page.goto('http://localhost:5173/login');
    await page.fill('input[type="email"]', 'juergenkrey@yahoo.de');
    await page.fill('input[type="password"]', 'start123'); // Annahme Standardpasswort
    await page.click('button:has-text("Anmelden")');
    await expect(page).toHaveURL(/.*\/admin/);

    // 2. VERANSTALTUNG ANLEGEN
    // Navigiert automatisch zum Tab 'veranstaltungen'
    await page.click('button:has-text("+ Neu")');
    await page.fill('label:has-text("Name der Veranstaltung") + input', 'Test Event 2025');
    await page.fill('label:has-text("Beginnt am") + input', '2025-05-20T09:00');
    await page.fill('label:has-text("Ort / Adresse") + input', 'Linz am Rhein');
    await page.click('button:has-text("Speichern")');
    await expect(page.locator('td:has-text("Test Event 2025")')).toBeVisible();

    // 3. VERANSTALTUNG AUSWÄHLEN
    await page.selectOption('select', { label: /Test Event 2025/ });
    // Warten, bis Daten geladen sind
    await page.waitForTimeout(1000); // Kurze Pause, um Ladevorgänge abzuwarten

    // 4. GEBÄUDE ANLEGEN
    await page.click('nav button:has-text("gebäude")');
    await page.click('button:has-text("+ Neu")');
    await page.fill('label:has-text("Name des Gebäudes") + input', 'Hauptgebäude A');
    await page.fill('label:has-text("Straße") + input', 'Schulstraße');
    await page.fill('label:has-text("Postleitzahl") + input', '53545');
    await page.fill('label:has-text("Ort") + input', 'Linz');
    await page.click('button:has-text("Speichern")');
    await expect(page.locator('td:has-text("Hauptgebäude A")')).toBeVisible();

    // 5. SLOTS ANLEGEN (3 Slots a 45min, 15min Pause)
    await page.click('nav button:has-text("slots")');

    const slots = [
      { name: 'Slot 1', start: '2025-05-20T09:00', end: '2025-05-20T09:45' },
      { name: 'Slot 2', start: '2025-05-20T10:00', end: '2025-05-20T10:45' },
      { name: 'Slot 3', start: '2025-05-20T11:00', end: '2025-05-20T11:45' }
    ];

    for (const slot of slots) {
      await page.click('button:has-text("+ Neu")');
      await page.fill('label:has-text("Bezeichnung") + input', slot.name);
      await page.fill('label:has-text("Beginn") + input', slot.start);
      await page.fill('label:has-text("Ende") + input', slot.end);
      await page.click('button:has-text("Slot erstellen")');
      await expect(page.locator(`td:has-text("${slot.name}")`)).toBeVisible(); // Bestätigen, dass der Slot sichtbar ist
    }

    // 6. REFERENTEN & VORTRÄGE ANLEGEN (Beispielhaft für einen)
    await page.click('nav button:has-text("benutzer")');
    await page.click('button:has-text("+ Neu")');
    await page.selectOption('select:has-text("Rolle")', 'REFERENT'); // Rolle auswählen
    await page.fill('label:has-text("Vorname") + input', 'Max');
    await page.fill('label:has-text("Nachname") + input', 'Referent');
    await page.fill('label:has-text("E-Mail") + input', 'max@referent.de');
    await page.click('button:has-text("Speichern")');
    await expect(page.locator('td:has-text("max@referent.de")')).toBeVisible(); // Referent in der Liste prüfen

    await page.click('nav button:has-text("vorträge")');
    await page.click('button:has-text("+ Neu")');
    await page.fill('label:has-text("Titel") + input', 'Einführung in Playwright');
    await page.selectOption('select:has-text("Referent")', { label: /Referent/ }); // Referent auswählen
    await page.click('button:has-text("Speichern")');
    await expect(page.locator('td:has-text("Einführung in Playwright")')).toBeVisible();

    // 7. TEILNEHMER ANLEGEN (Beispielhaft für einen)
    await page.click('nav button:has-text("benutzer")'); // Zurück zum Benutzer-Tab
    await page.click('button:has-text("+ Neu")');
    await page.fill('label:has-text("Vorname") + input', 'Tom');
    await page.fill('label:has-text("Nachname") + input', 'Teilnehmer');
    await page.fill('label:has-text("E-Mail") + input', 'tom@student.de');
    await page.click('button:has-text("Speichern")');
    await expect(page.locator('td:has-text("tom@student.de")')).toBeVisible();

    // 8. KONTROLLE IN VORTRAGSPLANUNG
    await page.click('nav button:has-text("planung")');
    await expect(page.locator('div:has-text("Teilnehmer") + div')).toContainText('1'); // Da wir nur 1 angelegt haben
    await expect(page.locator('div:has-text("Vorträge") + div')).toContainText('1');
    await expect(page.locator('div:has-text("Referenten") + div')).toContainText('1');
    await expect(page.locator('button:has-text("Jetzt Optimieren")')).toBeEnabled();
  });

});
----
